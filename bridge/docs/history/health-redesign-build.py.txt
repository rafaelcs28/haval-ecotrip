# ── sub-bloco: cada card antigo vira um <details> interno, corpo intacto ──────
def sub(k, label=None, open_default=False):
    c = cards[k]
    head = c['head'] or ('head-sub-' + k)
    return f'''        <details class="sub" id="sub-{k}" data-k="sub-{k}"{' open' if open_default else ''}>
          <summary><span class="grp">{label or c['title']}</span>
            <span class="sub-head" id="{head}">—</span><span class="chev">&rsaquo;</span></summary>
          <div class="sub-body">
{c['inner']}
          </div>
        </details>'''

def section(sid, name, order, dots, subs, extra_head='', open_attr=' open'):
    return f'''    <details class="sec" id="sec-{sid}" data-k="sec-{sid}" data-state="ok" style="order:{order}"{open_attr}>
      <summary>
        <span class="sq"></span>
        <span class="nm">{name}</span>
        <span class="sum" id="sec-{sid}-sum">—</span>
        <span class="dots" id="sec-{sid}-dots">{'<span class="dot"></span>' * dots}</span>
        <span class="age" id="sec-{sid}-age"></span>
        <span class="chev">&rsaquo;</span>
      </summary>
      <div class="body">
{extra_head}
{chr(10).join(subs)}
      </div>
    </details>'''

# ── script ───────────────────────────────────────────────────────────────────
# Era `old_js` + ~60 js.replace() em cadeia. Isso deu o que tinha que dar: cada
# correção nova em public/health.html exigia inventar um patch equivalente aqui, e
# dois patches encostando na mesma linha faziam o rebuild divergir em silêncio —
# o pior modo de falha possível pra um montador cuja única função é reproduzir o
# arquivo. E a derivação já não era reproduzível de todo jeito: `old_script.js` e
# `cards.json` moram no scratchpad da sessão do redesenho de 30/07.
#
# Agora o script é um snapshot versionado ao lado deste arquivo. A página segue
# sendo a fonte da verdade; pra atualizar o snapshot:
#     python3 -c "import io;p=io.open('public/health.html',encoding='utf-8').read();\
#     i=p.index('<script>');j=p.rindex('</script>')+9;\
#     io.open('docs/health.script.js','w',encoding='utf-8').write(p[i:j]+chr(10))"
js = io.open('/Users/consorciolimpagyn/haval-ecotrip/bridge/docs/health.script.js',
             encoding='utf-8').read().rstrip('\n')

js = js.replace('</body>', '').replace('</html>', '')
out = HEAD + '\n' + js + '\n</body>\n</html>\n'

# ── pós-processo do markup (revisão better-interface) ────────────────────────
# [4] Ilhas de hex cru vindas dos 21 cards antigos. No claro elas mediam 1,08:1
#     (#eee), 1,28:1 (#ffd966) e 1,49:1 (#e6b800) — duas delas em texto de ALERTA.
#     A troca é feita depois da montagem porque o hex vem de cards.json, não do JS.
_head, _body = out.split('</style>', 1)      # o bloco de tokens fica intacto
for _a, _b in [
    ('color:#888', 'color:var(--muted)'),
    ('border-bottom:1px solid #333', 'border-bottom:1px solid var(--line)'),
    ('border:1px solid #333', 'border:1px solid var(--line)'),
    ('background:#111', 'background:var(--bg)'),
    ('color:#eee', 'color:var(--text)'),
    ('color:#e6b800', 'color:var(--warn)'),
    ('color:#ffd966', 'color:var(--warn)'),
    ('color:#e74c3c', 'color:var(--crit)'),
    ('color:#7bc47f', 'color:var(--text)'),   # em cima de --ok-tint media 4,27:1
    ('background:#3a2a00', 'background:var(--warn-tint)'),
    ('border:1px solid #e6b800', 'border:1px solid var(--warn)'),
